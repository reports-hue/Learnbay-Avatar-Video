// New pipeline order:
async function generateVideo(params, sseEmit) {

  // Step 1: Research company (if brand settings exist)
  sseEmit('step', 'Researching company...');
  const companyIntel = params.useBrandSettings && params.brandSettings?.website
    ? await companyIntelligenceService.getCompanyIntel(params.brandSettings)
    : null;

  // Step 2: Generate AI background (parallel with script)
  sseEmit('step', 'Generating AI background...');
  const [bgPath, script] = await Promise.all([
    backgroundService.generateBackground({
      topic: params.topic,
      companyName: companyIntel?.companyName || params.brandSettings?.name,
      industry: params.brandSettings?.industry,
      platform: params.platform,
      brandColors: params.brandSettings?.colors || [],
      videoStyle: params.videoStyle
    }),
    openAIService.generateScript({
      topic: params.topic,
      platform: params.platform,
      style: params.scriptStyle,
      companyIntel,
      duration: getPlatformDuration(params.platform)
    })
  ]);

  // Step 3: Generate voice + word timings
  sseEmit('step', 'Generating voice...');
  const { audioPath, wordTimings } = 
    await speechService.synthesize(script, params.voice, params.voiceStyle);

  // Step 4: Request Azure Avatar (async, wait up to 8 min)
  sseEmit('step', 'Rendering avatar...');
  const avatarPath = await avatarService.renderAvatar({
    script,
    voice: params.voice,
    avatarCharacter: params.avatarCharacter,
    avatarStyle: params.avatarStyle,
    backgroundColor: '#00FF00FF' // green screen
  });

  // Step 5: Assemble final video
  sseEmit('step', 'Assembling video...');
  const videoPath = await ffmpegService.assemble({
    avatarPath,
    bgPath,
    audioPath,
    wordTimings,
    logoPath: params.logoPath || params.brandSettings?.logoUrl,
    ctaText: params.ctaText,
    platform: params.platform,
    accentColor: params.brandSettings?.accentColor || '#A855F7',
    videoId: params.videoId
  });

  // Step 6: Generate thumbnail
  sseEmit('step', 'Generating thumbnail...');
  const thumbnailPath = await generateThumbnail(videoPath, params.videoId);

  return { videoPath, thumbnailPath, script, companyIntel };
}

async function generateThumbnail(videoPath: string, videoId: string): Promise<string> {
  const thumbnailPath = `/tmp/thumb_${videoId}.jpg`;
  
  // Extract frame at 2.5 seconds (after avatar appears, good face shot):
  await execPromise(
    `${ffmpegBin} -ss 2.5 -i "${videoPath}" -vframes 1 ` +
    `-vf "scale=1080:1920:force_original_aspect_ratio=decrease,` +
    `pad=1080:1920:(ow-iw)/2:(oh-ih)/2" ` +
    `"${thumbnailPath}" -y`
  );
  
  return thumbnailPath;
}
